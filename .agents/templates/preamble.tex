% Reusable IB-style preamble for XeLaTeX equity research reports

% ============ 页面布局 ============

\setlength{\parindent}{2em}

  {\normalfont\Large\bfseries\color{navy}}
  {\thesection}{1em}{}
\titleformat{\subsection}
  {\normalfont\large\bfseries\color{steelgrey}}
  {\thesubsection}{1em}{}
