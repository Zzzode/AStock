    document.addEventListener("DOMContentLoaded", function () {
        const searchContainer = document.querySelector('.header_search-container');
        const searchIcon = document.querySelector('.header_search-icon.gray');
        const searchInput = document.querySelector('.header_search-input');
        const clearButton = document.querySelector('.header_search-clear');
        const searchOverlay = document.querySelector('.header_search-overlay');

        if (!searchContainer || !searchIcon || !searchInput || !clearButton || !searchOverlay) return;

        let placeholderIcon = null;

        function createPlaceholder() {
            if (!placeholderIcon) {
                placeholderIcon = searchIcon.cloneNode(true);
                placeholderIcon.style.visibility = 'hidden';
                placeholderIcon.style.pointerEvents = 'none';
                placeholderIcon.style.position = 'relative';
                placeholderIcon.style.zIndex = '1';
                placeholderIcon.classList.add('placeholder-icon');
                searchIcon.parentNode.insertBefore(placeholderIcon, searchIcon.nextSibling);
            }
        }

        function removePlaceholder() {
            if (placeholderIcon && placeholderIcon.parentNode) {
                placeholderIcon.parentNode.removeChild(placeholderIcon);
                placeholderIcon = null;
            }
        }

        function hideSearchIcon() {
            searchIcon.style.visibility = 'hidden';
            searchIcon.style.position = 'absolute';
            searchIcon.style.left = '-9999px';
        }

        function showSearchIcon() {
            searchIcon.style.visibility = 'visible';
            searchIcon.style.position = 'relative';
            searchIcon.style.left = 'auto';
        }

        /**
         * Manage body scroll lock based on search activation state.
         * Using both body and html for better cross-device compatibility.
         */
        const syncScrollLock = (isActive) => {
            if (isActive) {
                document.body.style.setProperty('overflow', 'hidden', 'important');
                document.documentElement.style.setProperty('overflow', 'hidden', 'important');
            } else {
                document.body.style.removeProperty('overflow');
                document.documentElement.style.removeProperty('overflow');
            }
        };

        // MutationObserver to handle state-dependent UI changes
        const observer = new MutationObserver((mutations) => {
            mutations.forEach((mutation) => {
                if (mutation.type === 'attributes' && mutation.attributeName === 'class') {
                    const isActive = searchContainer.classList.contains('active');
                    syncScrollLock(isActive);
                    if (!isActive) {
                        setTimeout(() => {
                            showSearchIcon();
                            removePlaceholder();
                        }, 100);
                    }
                }
            });
        });

        observer.observe(searchContainer, { attributes: true, attributeFilter: ['class'] });

        // Event Handlers
        searchIcon.addEventListener('click', (e) => {
            e.preventDefault();
            createPlaceholder();
            hideSearchIcon();
            searchContainer.classList.add('active');
            setTimeout(() => searchInput.focus(), 50); // Small delay for mobile keyboards
        });

        searchOverlay.addEventListener('click', (e) => {
            e.preventDefault();
            searchContainer.classList.remove('active');
        });

        document.addEventListener('click', (e) => {
            if (!searchContainer.contains(e.target) && !searchIcon.contains(e.target)) {
                searchContainer.classList.remove('active');
            }
        });

        searchContainer.addEventListener('click', (e) => {
            e.stopPropagation();
        });

        clearButton.addEventListener('mousedown', (e) => {
            e.preventDefault(); // Prevent input focus loss when clicking X
        });

        clearButton.addEventListener('click', (e) => {
            e.preventDefault();
            e.stopPropagation();
            searchInput.value = '';
            searchContainer.classList.remove('active');
            searchInput.blur();
        });

        // Search Execution
        const executeSearch = (query, src) => {
            if (!query) return;
            window.location.href = `/search?query=${encodeURIComponent(query)}&src=${encodeURIComponent(src)}`;
        };

        searchInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter') {
                executeSearch(searchInput.value.trim(), searchInput.getAttribute('data-search-bar') || 'navbar');
            }
        });

        document.querySelectorAll('[data-search-button]').forEach(button => {
            button.addEventListener('click', () => {
                const searchBar = button.getAttribute('data-search-button');
                const targetInput = document.querySelector(`[data-search-bar=${searchBar}]`);
                if (targetInput) executeSearch(targetInput.value.trim(), searchBar);
            });
        });

// 取得當前語系的購物車 cookie 名稱
window.getCartCookieName = function() {
    const locale = 'en';
    const languageMap = {
        'zh-tw': 'tw',
        // 專案需求：購物車僅需看 cart_tw / cart_en，將簡體語系對應到 en
        'zh-cn': 'en',
        'en': 'en'
    };
    const langSuffix = languageMap[locale] || 'en';
    // 使用 cart_{lang} 命名（cart_en, cart_tw, cart_cn）
    return 'cart_' + langSuffix;
};

// 全域函數：檢查商品是否已在購物車（立即定義，供其他腳本使用）
window.isInCart = function(packageId) {
        // 未登入用戶：檢查語系對應的 cart_{lang}
    const cartCookieName = window.getCartCookieName();
    const guestCart = window.getCookie(cartCookieName);
    if (guestCart) {
        try {
            const cartData = JSON.parse(guestCart);
            return cartData.some(function(item) { 
                return item.package_id == packageId;
            });
        } catch (e) {
            console.error('檢查購物車失敗:', e);
            return false;
        }
    }
    return false;
    };

// 全域輔助函數
function getCookie(name) {
    const value = '; ' + document.cookie;
    const parts = value.split('; ' + name + '=');
    if (parts.length === 2) {
        return decodeURIComponent(parts.pop().split(';').shift());
    }
    return null;
}

function setCookie(name, value, minutes) {
    if (minutes === 0) {
        // Session cookie（關閉瀏覽器後失效）
        document.cookie = name + '=' + encodeURIComponent(value) + ';path=/';
    } else {
        // 指定過期時間的 cookie
        const expires = new Date();
        expires.setTime(expires.getTime() + minutes * 60 * 1000);
        document.cookie = name + '=' + encodeURIComponent(value) + ';expires=' + expires.toUTCString() + ';path=/';
    }
}

// 合併購物車數據（去重，以 package_id 為 key）
function mergeCartData(existingItems, newItems) {
    const cartMap = new Map();
    existingItems.forEach(function(item) {
        if (item.package_id) cartMap.set(String(item.package_id), item);
    });
    newItems.forEach(function(item) {
        if (item.package_id && !cartMap.has(String(item.package_id))) {
            cartMap.set(String(item.package_id), item);
        }
    });
    return Array.from(cartMap.values());
}

function updateCartDisplay(cartData) {
    const itemsContainer = document.querySelector('.cart-dropdown-items');
    const countBadge = document.querySelector('.cart-count-badge');
    const totalAmount = document.querySelector('.cart-total-amount');
    const cartButton = document.querySelector('.cart-button');

    if (!cartData || cartData.length === 0) {
        if (cartButton) cartButton.classList.remove('has-items');
        if (itemsContainer) itemsContainer.innerHTML = ' Your cart is empty ';
                if (countBadge) countBadge.textContent = '0 items';
        if (totalAmount) totalAmount.textContent = 'USD 0';
                return;
    }
    if (cartButton) cartButton.classList.add('has-items');
        if (countBadge) countBadge.textContent = cartData.length + (cartData.length === 1 ? ' item' : ' items');
    
    let total = 0;
    let html = '';
    cartData.forEach(function(item) {
        const price = parseFloat(item.product_price || item.price) || 0;
        total += price;

        html += ' ';
        html += '     ';
        html += '         ' + (item.brand_product_name || item.package_name || item.name || '') + ' ';
                html += '         ' + item.currency + ' ' + price.toLocaleString('en-US') + ' ';
                html += '     ';
        html += '      ';
        html += ' ';
    });

    if (itemsContainer) itemsContainer.innerHTML = html;
        if (totalAmount) totalAmount.textContent = 'USD ' + total.toLocaleString('en-US');
    }

window.loadCartData = function() {
        // 未登入：從 Cookie 讀取（根據語系）
    const cartCookieName = getCartCookieName();
    const guestCart = getCookie(cartCookieName);
    
    if (guestCart) {
        try {
            const cartData = JSON.parse(guestCart);
            updateCartDisplay(cartData);
        } catch (e) {
            console.error('解析 Cookie 購物車失敗（可能是舊的加密格式）:', e);
            // 清除語系對應的 cart cookie
            document.cookie = cartCookieName + '=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
            updateCartDisplay([]);
        }
    } else {
        updateCartDisplay([]);
    }
    };

    function removeCookieItem(packageId) {
        // 1. 先呼叫後端 API 通知外部購物車系統
        fetch('/api/cart/removePackage', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content
            },
            body: JSON.stringify({ package_id: packageId })
        })
        .then(function(response) { return response.json(); })
        .then(function(data) {
            if (data.error === 0) {
                // 2. 更新本地 cookie
                const cartCookieName = getCartCookieName();
                const guestCart = getCookie(cartCookieName);
                if (guestCart) {
                    try {
                        let cartData = JSON.parse(guestCart);
                        cartData = cartData.filter(function(item) {
                            return item.package_id != packageId;
                        });
                        setCookie(cartCookieName, JSON.stringify(cartData), 525600); // 365天
                        updateCartDisplay(cartData);
                        
                        // 3. 復原該商品在頁面上的「加入購物車」按鈕狀態
                        restoreCartButton(packageId);
                    } catch (e) {
                        console.error('更新 Cookie 失敗:', e);
                    }
                }
            }
        })
        .catch(function(error) {
            console.error('移除失敗:', error);
        });
    }

            }

            }
        });
    }

    /**
     * 登入後合併語系對應的 guest_cart 到用戶購物車
     */
    function mergeGuestCartOnLogin() {
        const cartCookieName = getCartCookieName();
        const guestCart = getCookie(cartCookieName);
        if (!guestCart) {
            return;
        }

        try {
            const cartData = JSON.parse(guestCart);
            if (cartData.length === 0) {
                // 清除空的 cookie
                document.cookie = cartCookieName + '=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
                return;
            }

            // 逐一將商品加入用戶購物車
            let mergePromises = cartData.map(function(item) {
                return fetch('/api/cart/addPackage', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': '89DgWe8MyxG3Yp0Wk4wBnpuNS8ZnJurEcJbYWkTk'
                    },
                    body: JSON.stringify({
                        package_id: item.package_id,
                        report_number: item.report_number,
                        checked: 'true'
                    })
                })
                .then(function(response) { return response.json(); })
                .then(function(json) {
                    if (json.error === 0) {
                        return { success: true, item: item };
                    } else if (json.error === 2) {
                        return { success: true, item: item, duplicate: true };
                    } else {
                        return { success: false, item: item };
                    }
                })
                .catch(function(error) {
                    console.error('合併商品錯誤:', item.package_name, error);
                    return { success: false, item: item };
                });
            });

            // 等待所有商品合併完成
            Promise.all(mergePromises).then(function(results) {
                // 清除語系對應的 guest_cart cookie
                document.cookie = cartCookieName + '=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
                
                // 清除會員購物車 cookie，強制重新從 API 讀取
                const userId = '';
                if (userId) {
                    document.cookie = 'user_cart_' + userId + '=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
                }
                
                // 重新載入購物車顯示
                loadCartData();
            });

        } catch (e) {
            console.error('合併訪客購物車失敗:', e);
            // 清除可能損壞的 cookie
            document.cookie = cartCookieName + '=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;';
        }
    }

    function safeGetCookie(name){
        try{
            if (typeof window.getCookie === 'function') return window.getCookie(name);
            const v = document.cookie.match('(^|;)\\s*' + name + '\\s*=\\s*([^;]+)');
            return v ? decodeURIComponent(v.pop()) : null;
        } catch(e){ return null; }
    }

    function appendGuestIdToUrl(url){
        try{
            if (!url || url.indexOf('/api/cart/toList') === -1) return url;
            var guest = safeGetCookie('guest_id');
            if (!guest) return url;
            if (url.indexOf('guest_id=') !== -1) return url;
            var sep = url.indexOf('?') === -1 ? '?' : '&';
            return url + sep + 'guest_id=' + encodeURIComponent(guest);
        } catch(e){ return url; }
    }

    // 攔截點擊 a[href]，優先處理導向購物車的連結
    document.addEventListener('click', function(e){
        var a = e.target.closest && e.target.closest('a[href]');
        if (!a) return;
        var href = a.getAttribute('href');
        if (!href) return;
        if (href.indexOf('/api/cart/toList') === -1) return;

        var newHref = appendGuestIdToUrl(href);
        if (newHref === href) return; // 已包含或無 guest_id

        e.preventDefault();
        // 處理使用者希望在新分頁開啟的情況
        if (e.ctrlKey || e.metaKey || a.target === '_blank' || e.button === 1) {
            window.open(newHref, '_blank');
            return;
        }

        // 否則導向同一視窗
        window.location.href = newHref;
    }, true);

    // 包裝 window.open

  $(document).ready(function () {
    $('#btn-contact-inquery').on("click", function () {
      const csrfToken = document.querySelector('meta[name="csrf-token"]').getAttribute('content');
      const transError = {
        email: 'The E-mail Address must be a valid email address.',
        message: 'Message cannot be empty!',
      };

      const form = document.getElementById('contact-us-popup-form');
      let fromMail = form.from.value;
      let emailValid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(fromMail);
      if (!emailValid) {
        alert(transError.email);
        return false;
      }
      if (form.message.value.trim() === '' || /^\s*$/.test(form.message.value)) {
        alert(transError.message);
        return false;
      }

      $(this).attr('disabled', true);

      grecaptcha.ready(function () {
        grecaptcha.execute('6LeXsBYmAAAAACtiRX4fJHLQYceYn-RyZWrUcURv', { action: 'contactSales' }).then(function (token) {
          fetch('/api/contact/inquiry', {
            method: 'POST',
            headers: {
              'X-Requested-With': 'XMLHttpRequest',
              'Content-Type': 'application/json',
              'X-CSRF-TOKEN': csrfToken,
            },
            body: JSON.stringify({
              from: document.getElementById('contactFromEmail').value,
              message: document.getElementById('contactMessage').value,
              recaptcha: token,
              url: window.location.href
            }),
          })
            .then(res => res.json())
            .then(data => {
              console.log('json', data);
              $('#btn-contact-inquery').attr('disabled', false);
              if (data.error == 0) {
                $.magnificPopup.close();
              }
              alert(data.message);
            })
            .catch((error) => {
              console.log('error', error);
              alert('Error: sent failed');
              $('#btn-contact-inquery').attr('disabled', false);
            });
        });
      });
    });

    $('#btn-contact-close').on("click", function () {
      $.magnificPopup.close();
    });

    // 會員方案專用的 simple-inquiry 邏輯
    $('#btn-contact-membership-inquery').on("click", function () {
      const csrfToken = document.querySelector('meta[name="csrf-token"]').getAttribute('content');
      const transError = {
        email: 'The E-mail Address must be a valid email address.',
        message: 'Message cannot be empty!',
      };

      const form = document.getElementById('contact-us-membership-form');
      let emailVal = form.email.value;
      let emailValid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailVal);
      if (!emailValid) {
        alert(transError.email);
        return false;
      }
      if (form.message.value.trim() === '' || /^\s*$/.test(form.message.value)) {
        alert(transError.message);
        return false;
      }

      $(this).attr('disabled', true);

      fetch('/api/contact/simple-inquiry', {
        method: 'POST',
        headers: {
          'X-Requested-With': 'XMLHttpRequest',
          'Content-Type': 'application/json',
          'X-CSRF-TOKEN': csrfToken,
        },
        body: JSON.stringify({
          email: document.getElementById('contactFromEmailMembership').value,
          message: document.getElementById('contactMessageMembership').value,
          package_id: document.getElementById('contactPackageIdMembership').value || null
        }),
      })
        .then(res => res.json())
        .then(data => {
          console.log('json', data);
          $('#btn-contact-membership-inquery').attr('disabled', false);
          if (data.error == 0) {
            $.magnificPopup.close();
          }
          alert(data.message);
        })
        .catch((error) => {
          console.log('error', error);
          alert('Error: sent failed');
          $('#btn-contact-membership-inquery').attr('disabled', false);
        });
    });

    $('#btn-contact-membership-close').on("click", function () {
      $.magnificPopup.close();
      $('#contactPackageIdMembership').val('');
    });

    // 會員方案頁面：從按鈕取得 package_id 並填入隱藏欄位
    $(document).on('mousedown touchstart click', '[href="#contact-sales-membership"], .membership-contactus-btn', function (e) {
      var $el = $(this);
      var pkg = $el.data('package-id') || $el.attr('data-package-id') || $el.data('cart-package') || $el.attr('data-cart-package');
      if (!pkg) {
        var $a = $el.closest('a[href="#contact-sales-membership"]');
        pkg = $a.data('package-id') || $a.attr('data-package-id') || $a.data('cart-package') || $a.attr('data-cart-package');
      }
      $('#contactPackageIdMembership').val(pkg ? pkg : '');
    });
  });
